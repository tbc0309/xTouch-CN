cmake_minimum_required(VERSION 3.22)
include("${CMAKE_CURRENT_LIST_DIR}/../common/project_version.cmake")
if(CMAKE_HOST_WIN32)
 set(CMAKE_NINJA_FORCE_RESPONSE_FILE ON CACHE BOOL "Use response files on Windows" FORCE)
endif()
set(XTOUCH_BOARD "esp32-s3-touch-amoled-1.75")
set(IDF_TARGET "esp32s3")
include("${CMAKE_CURRENT_LIST_DIR}/../common/interactive_modules.cmake")
set(XTOUCH_DISPLAY_WIDTH 466)
set(XTOUCH_DISPLAY_HEIGHT 466)
set(XTOUCH_UI_NATIVE_CANVAS ON CACHE BOOL "Render the shared native-pixel canvas")
set(XTOUCH_UI_LAYOUT "round" CACHE STRING "Screen composition")
set(XTOUCH_ASSET_SIZE "large" CACHE STRING "Resource size")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/../sdkconfig.defaults;${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
option(XTOUCH_FEATURE_XIAOMI_BLE "Include Xiaomi BLE thermometer" ON)
if(XTOUCH_FEATURE_XIAOMI_BLE)
 if(XTOUCH_FEATURE_BLE_PROVISIONING)
  message(FATAL_ERROR "Xiaomi BLE and BLE provisioning are mutually exclusive")
 endif()
 list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/../common/sdkconfig.xiaomi_ble.defaults")
elseif(XTOUCH_FEATURE_BLE_PROVISIONING)
 list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/../common/sdkconfig.ble_provisioning.defaults")
endif()
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../main" "${CMAKE_CURRENT_LIST_DIR}/../.deps/lvgl" "${CMAKE_CURRENT_LIST_DIR}/xtouch_board_support")
if(XTOUCH_FEATURE_PIXEL_SCREEN)
 if(XTOUCH_FEATURE_BLE_PROVISIONING)
  message(FATAL_ERROR "Pixel screen and BLE provisioning are mutually exclusive")
 endif()
 list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/../common/sdkconfig.pixel_screen.defaults")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../common/idf_project.cmake")
set(COMPONENTS main)
project(xtouch_amoled175)
